package com.formation.controller;

import com.formation.entity.Course;
import com.formation.service.CourseService;
import jakarta.validation.Valid;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.validation.BindingResult;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.servlet.mvc.support.RedirectAttributes;

import java.util.List;

@Controller
@RequestMapping("/courses")
public class CourseController {

    @Autowired
    private CourseService courseService;

    @GetMapping
    public String listCourses(Model model) {
        List<Course> courses = courseService.getAllCourses();
        model.addAttribute("courses", courses);
        model.addAttribute("pageTitle", "Gestion des Cours");
        return "courses/list";
    }

    @GetMapping("/new")
    public String showCreateForm(Model model) {
        model.addAttribute("course", new Course());
        model.addAttribute("pageTitle", "Ajouter un nouveau cours");
        return "courses/form";
    }

    @PostMapping("/save")
    public String saveCourse(@Valid @ModelAttribute Course course,
                             BindingResult result,
                             Model model,
                             RedirectAttributes redirectAttributes) {

        if (result.hasErrors()) {
            model.addAttribute("pageTitle", "Ajouter un nouveau cours");
            return "courses/form";
        }

        if (!courseService.isCodeUnique(course.getCode())) {
            result.rejectValue("code", "error.course", "Ce code existe déjà!");
            model.addAttribute("pageTitle", "Ajouter un nouveau cours");
            return "courses/form";
        }

        courseService.saveCourse(course);
        redirectAttributes.addFlashAttribute("successMessage", "Cours ajouté avec succès!");
        return "redirect:/courses";
    }

    @GetMapping("/edit/{id}")
    public String showEditForm(@PathVariable Long id, Model model,
                               RedirectAttributes redirectAttributes) {
        try {
            Course course = courseService.getCourseById(id);
            model.addAttribute("course", course);
            model.addAttribute("pageTitle", "Modifier le cours: " + course.getNom());
            return "courses/form";
        } catch (RuntimeException e) {
            redirectAttributes.addFlashAttribute("errorMessage", e.getMessage());
            return "redirect:/courses";
        }
    }

    @PostMapping("/update/{id}")
    public String updateCourse(@PathVariable Long id,
                               @Valid @ModelAttribute Course course,
                               BindingResult result,
                               Model model,
                               RedirectAttributes redirectAttributes) {

        if (result.hasErrors()) {
            model.addAttribute("pageTitle", "Modifier le cours");
            return "courses/form";
        }

        Course existingCourse = courseService.getCourseById(id);

        if (!existingCourse.getCode().equals(course.getCode()) &&
                !courseService.isCodeUnique(course.getCode())) {
            result.rejectValue("code", "error.course", "Ce code existe déjà!");
            model.addAttribute("pageTitle", "Modifier le cours");
            return "courses/form";
        }

        courseService.updateCourse(id, course);
        redirectAttributes.addFlashAttribute("successMessage", "Cours modifié avec succès!");
        return "redirect:/courses";
    }

    @GetMapping("/delete/{id}")
    public String deleteCourse(@PathVariable Long id,
                               RedirectAttributes redirectAttributes) {
        try {
            courseService.deleteCourse(id);
            redirectAttributes.addFlashAttribute("successMessage", "Cours supprimé avec succès!");
        } catch (RuntimeException e) {
            redirectAttributes.addFlashAttribute("errorMessage", "Erreur lors de la suppression: " + e.getMessage());
        }
        return "redirect:/courses";
    }
}

        model.addAttribute("courses", courses);
        model.addAttribute("niveaux", niveaux);
        model.addAttribute("pageTitle", "Associer un cours à un niveau");
        return "course-niveaux/form";

                                   BindingResult result,
                                   Model model,

        if (result.hasErrors()) {
            List<Course> courses = courseService.getAllCourses();
            List<Niveau> niveaux = niveauService.getAllNiveaux();
            model.addAttribute("courses", courses);
            model.addAttribute("niveaux", niveaux);
            model.addAttribute("pageTitle", "Associer un cours à un niveau");
            return "course-niveaux/form";
        }

        // Check if association already exists
        if (courseNiveauService.existsByCourseAndNiveau(courseNiveau.getCourse(), courseNiveau.getNiveau())) {
            result.rejectValue("course", "error.courseNiveau", "Cette association existe déjà!");
            List<Course> courses = courseService.getAllCourses();
            List<Niveau> niveaux = niveauService.getAllNiveaux();
            model.addAttribute("courses", courses);
            model.addAttribute("niveaux", niveaux);
            model.addAttribute("pageTitle", "Associer un cours à un niveau");
            return "course-niveaux/form";
        }

        courseNiveauService.saveCourseNiveau(courseNiveau);
        redirectAttributes.addFlashAttribute("successMessage", "Association ajoutée avec succès!");
        return "redirect:/course-niveaux";
    }

    @GetMapping("/edit/{id}")
    public String showEditForm(@PathVariable Long id, Model model,
                               RedirectAttributes redirectAttributes) {
        try {
            CourseNiveau courseNiveau = courseNiveauService.getCourseNiveauById(id);
            List<Course> courses = courseService.getAllCourses();
            List<Niveau> niveaux = niveauService.getAllNiveaux();

            model.addAttribute("courseNiveau", courseNiveau);
            model.addAttribute("courses", courses);
            model.addAttribute("niveaux", niveaux);
            model.addAttribute("pageTitle", "Modifier l'association Cours-Niveau");
            return "course-niveaux/form";

            redirectAttributes.addFlashAttribute("errorMessage", e.getMessage());
            return "redirect:/course-niveaux";
        }
    }

    @PostMapping("/update/{id}")
    public String updateCourseNiveau(@PathVariable Long id,
                                     @Valid @ModelAttribute CourseNiveau courseNiveau,
                                     BindingResult result,
                                     Model model,
                                     RedirectAttributes redirectAttributes) {

        if (result.hasErrors()) {
            List<Course> courses = courseService.getAllCourses();
            List<Niveau> niveaux = niveauService.getAllNiveaux();
            model.addAttribute("courses", courses);
            model.addAttribute("niveaux", niveaux);
            model.addAttribute("pageTitle", "Modifier l'association Cours-Niveau");
            return "course-niveaux/form";
        }

        CourseNiveau existingCourseNiveau = courseNiveauService.getCourseNiveauById(id);

        // Check if new association already exists (excluding current)
        if (!existingCourseNiveau.getCourse().equals(courseNiveau.getCourse()) ||
                !existingCourseNiveau.getNiveau().equals(courseNiveau.getNiveau())) {
            if (courseNiveauService.existsByCourseAndNiveau(courseNiveau.getCourse(), courseNiveau.getNiveau())) {
                result.rejectValue("course", "error.courseNiveau", "Cette association existe déjà!");
                List<Course> courses = courseService.getAllCourses();
                List<Niveau> niveaux = niveauService.getAllNiveaux();
                model.addAttribute("courses", courses);
                model.addAttribute("niveaux", niveaux);
                model.addAttribute("pageTitle", "Modifier l'association Cours-Niveau");
                return "course-niveaux/form";
            }
        }

        courseNiveauService.updateCourseNiveau(id, courseNiveau);
        redirectAttributes.addFlashAttribute("successMessage", "Association modifiée avec succès!");
        return "redirect:/course-niveaux";
    }

    @GetMapping("/delete/{id}")
    public String deleteCourseNiveau(@PathVariable Long id,

            redirectAttributes.addFlashAttribute("successMessage", "Association supprimée avec succès!");
        } catch (RuntimeException e) {
            redirectAttributes.addFlashAttribute("errorMessage", "Erreur lors de la suppression: " + e.getMessage());

            return "students/form";

        if (!existingStudent.getMatricule().equals(student.getMatricule()) &&
                !studentService.isMatriculeUnique(student.getMatricule())) {
            result.rejectValue("matricule", "error.student", "Ce matricule existe déjà!");
            model.addAttribute("pageTitle", "Modifier l'étudiant");
            return "students/form";
        }

        if (!existingStudent.getEmail().equals(student.getEmail()) &&
                !studentService.isEmailUnique(student.getEmail())) {
            result.rejectValue("email", "error.student", "Cet email existe déjà!");
            model.addAttribute("pageTitle", "Modifier l'étudiant");
            return "students/form";
        }

        studentService.updateStudent(id, student);

    document.addEventListener('DOMContentLoaded', function() {
        // Smooth scrolling for anchor links
        document.querySelectorAll('a[href^="#"]').forEach(anchor => {
            anchor.addEventListener('click', function (e) {
                e.preventDefault();
                const targetId = this.getAttribute('href');
                if (targetId === '#') return;

                const targetElement = document.querySelector(targetId);
                if (targetElement) {
                    window.scrollTo({
                        top: targetElement.offsetTop - 20,
                        behavior: 'smooth'
                    });
                }
            });
        });

        // Update current year in footer
        const currentYear = new Date().getFullYear();
        const yearElement = document.querySelector('.fa-copyright').parentElement;
        if (yearElement.textContent.includes('2025')) {
            yearElement.innerHTML = yearElement.innerHTML.replace('2025', currentYear);
        }
    });
</script>
